import java.util.*;
import com.google.gson.*;
import com.pixelmonmod.pixelmon.client.models.items.PokeBallItemModel;
import com.pixelmonmod.pixelmon.client.models.items.PokeBallLidItemModel;
import local.pixelmon.backport.*;
import local.pixelmon.backport.client.*;
import net.minecraft.util.ResourceLocation;
import net.minecraft.client.renderer.model.*;
import net.minecraftforge.client.event.ModelBakeEvent;

/** Update legacy advancement icons without replacing the shared NBT ball wrappers. */
public final class BallArt170Cases {
    static void check(boolean ok,String why){if(!ok)throw new AssertionError(why);}
    static IBakedModel model(){return new IBakedModel(){};}
    public static void run()throws Exception{
        Map<ResourceLocation,IBakedModel> models=new HashMap<>();
        IBakedModel shared=model(),lid=model();
        models.put(new ModelResourceLocation("pixelmon:poke_ball","inventory"),shared);
        models.put(new ModelResourceLocation("pixelmon:poke_ball_lid","inventory"),lid);
        for(String name:BallItemArt.FILES){
            models.put(new ModelResourceLocation("pixelmon:"+name,"inventory"),model());
            models.put(new ResourceLocation(Content.NS,"item/ball_parts/"+name),model());
            JsonObject json=Content.read("assets/pixelmon_backport/models/item/ball_parts/"+name+".json");
            check(json.getAsJsonObject("textures").get("layer0").getAsString().equals(Content.NS+":items/pokeballs/"+name),"Old static ball art: "+name);
        }
        BallItemArtModels.bake(new ModelBakeEvent(null,models,null));
        for(String name:BallItemArt.FILES)if(!name.equals("poke_ball")&&!name.equals("poke_ball_lid"))
            check(models.get(new ModelResourceLocation("pixelmon:"+name,"inventory"))==models.get(new ResourceLocation(Content.NS,"item/ball_parts/"+name)),"Legacy ball model unchanged: "+name);
        check(models.get(new ModelResourceLocation("pixelmon:poke_ball","inventory")) instanceof PokeBallItemModel,"Shared ball NBT wrapper lost");
        check(models.get(new ModelResourceLocation("pixelmon:poke_ball_lid","inventory")) instanceof PokeBallLidItemModel,"Shared lid NBT wrapper lost");
        check(((PokeBallItemModel)models.get(new ModelResourceLocation("pixelmon:poke_ball","inventory"))).model==models.get(new ResourceLocation(Content.NS,"item/ball_parts/poke_ball")),"Shared fallback art old");
        check(((PokeBallLidItemModel)models.get(new ModelResourceLocation("pixelmon:poke_ball_lid","inventory"))).model==models.get(new ResourceLocation(Content.NS,"item/ball_parts/poke_ball_lid")),"Shared lid fallback art old");
        for(int i=0;i<100;i++)check("b站端忆制作".equals(TitleMessages.next()),"Splash must remain the author credit");
        for(var e:Content.read("backport/content.json").getAsJsonObject("referenceExtras").getAsJsonObject("drops/pokechestdrops").entrySet())
            check(!e.getValue().toString().contains("pixelmon_backport:metal_alloy"),"Alloy remains in chest pool");
        check(Content.read("backport/spawn-npc-names.json").get("OldFisherman").getAsString().equals("老渔夫"),"Fisherman rename");
        System.out.println("PASS production model-bake handler with isolated render registry: owned legacy ball/lid/part models updated, shared NBT wrappers retained; splash choices and alloy chest source removal. No game launched.");
    }
}
